## Zona horaria: el mismo instante cae en meses distintos
SELECT current_setting('TimeZone') AS zona,
        date_trunc('month', timestamptz '2026-02-28 23:30:00-03')::date AS mes
--> [
  {
    "zona": "America/Argentina/Buenos_Aires",
    "mes": "2026-02-01T03:00:00.000Z"
  }
]

## Mismo instante con TimeZone UTC
SELECT current_setting('TimeZone') AS zona,

--> [

    "zona": "UTC",
    "mes": "2026-03-01T03:00:00.000Z"
